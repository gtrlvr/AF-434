// iCL42-08 keyboard control
// Wiring: Pin 9 -> PUL+, Pin 8 -> DIR+, PUL- and DIR- -> Arduino GND
// Serial commands (sent by the Python script):
//   'a' = run counter-clockwise (A key held)
//   'd' = run clockwise         (D key held)
//   's' = stop                  (key released)
//   'q' = speed up
//   'e' = speed down

const int PUL_PIN = 9;
const int DIR_PIN = 8;

bool running = false;
bool currentDirCW = true;

// Step interval in microseconds (time between pulses).
// 500 us = 2000 Hz. At the driver's default 1600 pulses/rev that's ~75 RPM.
unsigned long stepIntervalUs = 500;
const unsigned long MIN_INTERVAL = 60;    // ~16.6 kHz max
const unsigned long MAX_INTERVAL = 5000;  // ~200 Hz min

unsigned long lastStepUs = 0;

void setup() {
  pinMode(PUL_PIN, OUTPUT);
  pinMode(DIR_PIN, OUTPUT);
  digitalWrite(PUL_PIN, LOW);
  digitalWrite(DIR_PIN, LOW);
  Serial.begin(115200);
}

void setDirection(bool cw) {
  if (cw != currentDirCW) {
    running = false;                 // brief pause so DIR settles before pulses
    digitalWrite(cw ? DIR_PIN : DIR_PIN, cw ? HIGH : LOW);
    digitalWrite(DIR_PIN, cw ? HIGH : LOW);
    delayMicroseconds(10);           // iCL42 needs DIR stable ~5us before PUL
    currentDirCW = cw;
  }
}

void loop() {
  // Handle incoming commands
  while (Serial.available()) {
    char c = Serial.read();
    switch (c) {
      case 'd':                      // clockwise
        setDirection(true);
        running = true;
        break;
      case 'a':                      // counter-clockwise
        setDirection(false);
        running = true;
        break;
      case 's':
        running = false;
        break;
      case 'q':                      // speed up (shorter interval)
        stepIntervalUs = stepIntervalUs * 8 / 10;
        if (stepIntervalUs < MIN_INTERVAL) stepIntervalUs = MIN_INTERVAL;
        break;
      case 'e':                      // speed down (longer interval)
        stepIntervalUs = stepIntervalUs * 10 / 8;
        if (stepIntervalUs > MAX_INTERVAL) stepIntervalUs = MAX_INTERVAL;
        break;
    }
  }

  // Non-blocking pulse generation
  if (running && (micros() - lastStepUs >= stepIntervalUs)) {
    lastStepUs = micros();
    digitalWrite(PUL_PIN, HIGH);
    delayMicroseconds(5);            // pulse width, above the 2.5us minimum
    digitalWrite(PUL_PIN, LOW);
  }
}
